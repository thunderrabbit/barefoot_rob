{{- /* The values every calendar target needs, derived once from front matter. Takes a Page, returns a dict. Callers must check EventDate is set first. */ -}} {{- $page := . -}} {{- $start := time.AsTime $page.Params.EventDate -}} {{- /* An event with no end would be zero-length or all-day depending on the calendar app, so fall back to an hour. */ -}} {{- $end := $start.Add (time.ParseDuration "1h") -}} {{- with $page.Params.EventEndDate -}} {{- $end = time.AsTime . -}} {{- end -}} {{- /* EventLocation is the human label, EventLocationURL the Zoom/Meet/Maps link behind it. Phone calendars linkify the URL in the WHERE field. */ -}} {{- $location := printf "%v" ($page.Params.EventLocation | default "") -}} {{- with $page.Params.EventLocationURL -}} {{- if eq $location "" -}} {{- $location = printf "%v" . -}} {{- else -}} {{- $location = printf "%s — %v" $location . -}} {{- end -}} {{- end -}} {{- /* Send people back to the site for official details. */ -}} {{- $details := $page.Permalink -}} {{- with $page.Params.TimeDescription -}} {{- $details = printf "%v\n\n%s" . $page.Permalink -}} {{- end -}} {{- /* The name the .ics lands under in a Downloads folder. Hugo's baseName is per output format, not per page, so every event's calendar file is served as index.ics; only an can give it a useful name. Dated from EventDate, not .Date, because .Date is when the file was written -- and because two Ring Ring sessions share a title, so without the date the second one saves as "... (1).ics". anchorize rather than urlize: urlize keeps the period in "Calling." and percent-encodes a Japanese title into a filename nobody can read. */ -}} {{- /* Trim is NOT piped: a pipeline appends the value as the LAST argument, so `| strings.Trim "-"` trims the slug's characters out of "-" and hands back an empty string. */ -}} {{- $slug := $page.Title | anchorize | replaceRE `-{2,}` "-" -}} {{- $slug = strings.Trim $slug "-" -}} {{- $filename := printf "%s-%s.ics" ($start.Format "2006-01-02") $slug -}} {{- return dict "start" $start "end" $end "location" $location "details" $details "filename" $filename -}}
